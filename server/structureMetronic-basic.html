📋 2. Data Structure (employee_models.ts)

export interface EmployeeData {
  id?: string
  employee_code: string        // รหัสพนักงาน
  first_name_th: string        // ชื่อไทย
  last_name_th: string         // นามสกุลไทย
  first_name_en: string        // ชื่ออังกฤษ
  last_name_en: string         // นามสกุลอังกฤษ
  nickname: string             // ชื่อเล่น
  email: string                // อีเมล
  phone: string                // เบอร์โทร
  image_url: string            // รูปโปรไฟล์
  department_id: string        // id แผนก
  position_id: string          // id ตำแหน่ง
  hire_date: string            // วันที่เริ่มงาน
  salary: number               // เงินเดือน
  status: 'Active' | 'Inactive' | 'On Leave'
  createdAt?: string
  updatedAt?: string
}
